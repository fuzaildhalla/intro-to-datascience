Python Program: Bar Chart of Movies by Genre

import pandas as pd
import matplotlib.pyplot as plt

# Create a small movie dataset
data = {
    "movie_name": [
        "Dangal", "3 Idiots", "Pathaan", "Jawan",
        "Stree", "Bhool Bhulaiyaa", "KGF", "RRR",
        "Drishyam", "Andhadhun"
    ],
    "genre": [
        "Drama", "Drama", "Action", "Action",
        "Comedy", "Comedy", "Action", "Action",
        "Thriller", "Thriller"
    ]
}

df = pd.DataFrame(data)

# Count movies in each genre
genre_count = df["genre"].value_counts()

# Create a simple bar chart
plt.bar(genre_count.index, genre_count.values)
plt.xlabel("Genre")
plt.ylabel("Number of Movies")
plt.title("Number of Movies by Genre")
plt.show()
